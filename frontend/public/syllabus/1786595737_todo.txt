# To-Do List Management System

tasks = []


# Function to add task
def add_task():
    task = input("Enter task: ")

    tasks.append(task)

    print("Task added successfully!")


# Function to view tasks
def view_tasks():

    if len(tasks) == 0:
        print("No tasks found!")
        return

    print("\n========== TASKS ==========")

    for task in tasks:
        print(task)


# Function to save tasks
def save_tasks():

    if len(tasks) == 0:
        print("No tasks to save!")
        return

    with open("todo.txt", "a") as file:

        for task in tasks:
            file.write(task + "\n")

    print("Tasks saved successfully!")


# Main menu
while True:

    print("\n1. Add Task")
    print("2. View Tasks")
    print("3. Save Tasks")
    print("4. Exit")

    choice = input("Enter your choice: ")

    if choice == "1":
        add_task()

    elif choice == "2":
        view_tasks()

    elif choice == "3":
        save_tasks()

    elif choice == "4":
        print("Thank you!")
        break

    else:
        print("Invalid choice!")
