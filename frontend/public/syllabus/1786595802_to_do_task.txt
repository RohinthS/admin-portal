while True:

    print("--------- To-Do List ---------")
    print("1. Add Task")
    print("2. View Task")
    print("3. Complete Task")
    print("4. Exit")

    choice = int(input("Enter your choice: "))

    if choice == 1:

        task = input("Enter your task: ")

        file = open("to do list.txt", "a")
        file.write(task + "\n")
        file.close()

        print("Task added successfully")

    elif choice == 2:

        file = open("to do list.txt", "r")
        data = file.read()
        print(data)
        file.close()

    elif choice == 3:

        file = open("to do list.txt", "r")
        complete = file.readlines()
        file.close()

        for i in range(len(complete)):
            print(i + 1, complete[i].strip())

        number = int(input("Enter task number: "))

        complete[number - 1] = complete[number - 1].strip() + " - Completed\n"

        file = open("to do list.txt", "w")
        file.writelines(complete)
        file.close()

        print("Task completed successfully")

    elif choice == 4:

        print("Thank you!")
        break

    else:

        print("Invalid choice!")
