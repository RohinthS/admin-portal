

FILE_NAME = "todo.txt"


def load_tasks():
    tasks = []

    try:
        file = open(FILE_NAME, "r")

        for line in file:
            task = line.strip()

            if task != "":
                tasks.append(task)

        file.close()

    except FileNotFoundError:
       
        file = open(FILE_NAME, "w")
        file.close()

    return tasks


# Save all tasks to file
def save_tasks(tasks):
    file = open(FILE_NAME, "w")

    for task in tasks:
        file.write(task + "\n")

    file.close()


# Display tasks
def show_tasks(tasks):
    if len(tasks) == 0:
        print("\nNo tasks found.")
        return

    print("\n----- YOUR TO-DO LIST -----")

    for i in range(len(tasks)):
        print(f"{i + 1}. {tasks[i]}")

    print("---------------------------")


# Add a new task
def add_task(tasks):
    task = input("\nEnter your task: ")

    if task.strip() == "":
        print("Task cannot be empty.")
        return

    tasks.append(task)
    save_tasks(tasks)

    print("Task added successfully!")


# Delete a task
def delete_task(tasks):
    show_tasks(tasks)

    if len(tasks) == 0:
        return

    try:
        number = int(input("\nEnter task number to delete: "))

        if number >= 1 and number <= len(tasks):
            removed_task = tasks.pop(number - 1)

            save_tasks(tasks)

            print(f"Task deleted: {removed_task}")

        else:
            print("Invalid task number.")

    except ValueError:
        print("Please enter a valid number.")


# Main program
tasks = load_tasks()

while True:

    print("\n===== TO-DO LIST APPLICATION =====")
    print("1. Add Task")
    print("2. View Tasks")
    print("3. Delete Task")
    print("4. Exit")
    print("==================================")

    choice = input("Enter your choice: ")

    if choice == "1":
        add_task(tasks)

    elif choice == "2":
        show_tasks(tasks)

    elif choice == "3":
        delete_task(tasks)

    elif choice == "4":
        print("\nThank you for using To-Do List!")
        break

    else:
        print("Invalid choice. Please select 1-4.")
